## `models/vehicle.py` — Задания 2–5, 8

```python
from abc import ABC, abstractmethod


class Vehicle(ABC):
    """Базовый класс транспортного средства."""

    def __init__(self, brand: str, model: str, year: int):
        self.brand = brand
        self.model = model
        self.year = year
        self._speed = 0          # защищённый атрибут (Задание 2)

    # ---------- Задание 4: инкапсуляция через property ----------
    @property
    def speed(self) -> int:
        return self._speed

    @speed.setter
    def speed(self, value: int):
        if value < 0:
            raise ValueError("Скорость не может быть отрицательной")
        self._speed = value

    # ---------- Задание 3 + 4: методы используют property ----------
    def accelerate(self, value: int):
        self.speed = self.speed + value   # через setter

    def brake(self, value: int):
        self.speed = max(0, self.speed - value)  # не ниже нуля

    def stop(self):
        self.speed = 0

    def get_info(self) -> str:
        return (f"{self.brand} {self.model} ({self.year}), "
                f"скорость: {self.speed} км/ч")

    # ---------- Задание 5: __str__ ----------
    def __str__(self) -> str:
        return (f"{self.brand} {self.model}, {self.year} г., "
                f"{self.speed} км/ч")

    # ---------- Задание 8: абстрактный метод ----------
    @abstractmethod
    def move(self) -> str:
        """Каждый тип ТС реализует собственный способ движения."""
        ...
```

**Пояснение к Заданию 8 (п. 5):** после добавления `@abstractmethod` класс `Vehicle` наследуется от `ABC`, и Python помечает его как абстрактный. При попытке `Vehicle(...)` интерпретатор выбросит `TypeError: Can't instantiate abstract class Vehicle with abstract method move`, потому что абстрактный метод не реализован. Это защищает от создания «недоделанных» объектов.

## `garage.py` — Задание 10

```python
from models.vehicle import Vehicle


class Garage:
    """Гараж работает только через интерфейс Vehicle,
    без isinstance() и проверок конкретных типов."""

    def __init__(self):
        self.vehicles: list[Vehicle] = []

    def add_vehicle(self, vehicle: Vehicle):
        if not isinstance(vehicle, Vehicle):
            raise TypeError("В гараж можно добавлять только Vehicle")
        self.vehicles.append(vehicle)

    def remove_vehicle(self, vehicle: Vehicle):
        if vehicle in self.vehicles:
            self.vehicles.remove(vehicle)

    def show_all(self):
        if not self.vehicles:
            print("Гараж пуст.")
            return
        print("=== Содержимое гаража ===")
        for i, v in enumerate(self.vehicles, 1):
            print(f"{i}. {v.get_info()}")

    def find_by_brand(self, brand: str) -> list[Vehicle]:
        return [v for v in self.vehicles
                if v.brand.lower() == brand.lower()]
```

**Пояснение к п. 6:** `Garage` не знает про `Car`, `Truck`, `Motorcycle` — он работает через общий интерфейс `Vehicle` (`get_info`, `brand`). Это позволяет добавлять новые классы без изменения `Garage`.

from models import Vehicle, Car, Truck, Motorcycle, ElectricCar
from garage import Garage


def main():

    try:
        v = Vehicle("Generic", "X", 2020)
    except TypeError as e:
        print(f"Ожидаемая ошибка: {e}")
    print()

    # ---------- Задание 6: объект Car ----------
    print("--- Задание 6: Car ---")
    car = Car("Toyota", "Camry", 2020, doors=4)
    print(car.get_info())
    car.accelerate(60)
    print(car.get_info())
    car.brake(100)        # скорость не станет отрицательной
    print(car.get_info())
    car.stop()
    print(car.get_info())
    print(car)            # __str__
    print()

    # ---------- Задание 4: обработка ValueError ----------
    print("--- Задание 4: попытка задать отрицательную скорость ---")
    try:
        car.speed = -10
    except ValueError as e:
        print(f"Ожидаемая ошибка: {e}")
    print()

    # ---------- Задания 9 и 12: полиморфизм ----------
    truck = Truck("Volvo", "FH16", 2019, max_load=20.0)
    moto = Motorcycle("Harley-Davidson", "Sportster", 2021, has_sidecar=False)
    ev = ElectricCar("Tesla", "Model 3", 2023, doors=4,
                     battery_kwh=75.0, range_km=560)

    vehicles = [car, truck, moto, ev]

    print("--- Задание 9: полиморфный вызов move() и get_info() ---")
    for v in vehicles:
        # никаких isinstance() — Python сам выбирает реализацию
        v.accelerate(50)
        print(v.move())
        print(v.get_info())

    print("--- Задание 10: Garage ---")
    garage = Garage()
    for v in vehicles:
        garage.add_vehicle(v)

    garage.show_all()
    print()

    print("--- Поиск по марке 'Tesla' ---")
    for v in garage.find_by_brand("Tesla"):
        print(v.get_info())
    print()

    # ---------- Задание 12: удаление и проверка ----------
    print("--- Удаление Truck из гаража ---")
    garage.remove_vehicle(truck)
    garage.show_all()


if __name__ == "__main__":
    main()

## Ответы на теоретические вопросы

### Задание 8, п. 5 — почему `Vehicle` нельзя создать?
`Vehicle` наследует `ABC` и содержит метод с `@abstractmethod`. Python при создании экземпляра проверяет, что все абстрактные методы переопределены в конкретном классе. Если есть хотя бы один нереализованный — выбрасывается `TypeError`. Это гарантирует, что объекты «абстрактного» класса не появятся.

### Задание 9, п. 5 — как Python выбирает реализацию?
При вызове `v.move()` Python смотрит на **фактический тип объекта** (`v.__class__`), а не на тип переменной. Сначала ищется метод в классе объекта, затем по цепочке MRO (Method Resolution Order) — у родителей. Так как `Car`, `Truck`, `Motorcycle`, `ElectricCar` переопределяют `move()` и `get_info()`, вызывается именно их версия. Это **динамический полиморфизм** (утиная типизация + наследование).

### Задание 12 — почему новый класс не требует правок?
`Garage` и цикл в `main()` работают только через методы `add_vehicle`, `get_info`, `move`, `brand` — всё это есть в интерфейсе `Vehicle`. `ElectricCar` соблюдает тот же контракт, поэтому подставляется в общий список без `isinstance`-проверок и без изменения `Garage` — принцип открытости/закрытости (OCP) соблюдён.
